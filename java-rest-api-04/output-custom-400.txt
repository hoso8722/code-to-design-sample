# 題名が空、冊数が範囲外
POST /  {"title":"","copies":-5}
  -> 400  {"message":"入力に誤りがあります","errors":[{"field":"copies","message":"冊数は0以上にしてください"},{"field":"title","message":"題名は必須です"}]}

# 題名が121文字
POST /  {"title":"（「あ」を121文字）","copies":1}
  -> 400  {"message":"入力に誤りがあります","errors":[{"field":"title","message":"題名は120文字以内にしてください"}]}

# 冊数が上限を超える
POST /  {"title":"こころ","copies":2000}
  -> 400  {"message":"入力に誤りがあります","errors":[{"field":"copies","message":"冊数は1000以下にしてください"}]}

# 形式は正しいが、題名が重複(業務ルール)
POST /  {"title":"坊っちゃん","copies":1}
  -> 409

# 冊数が数字でない(JSONの型が合わない)
POST /  {"title":"こころ","copies":"たくさん"}
  -> 400  {"timestamp":"2026-10-10T20:21:22.625Z","status":400,"error":"Bad Request","path":"/api/books"}

# PATCH:冊数だけが範囲外
PATCH /1  {"copies":-1}
  -> 400  {"message":"入力に誤りがあります","errors":[{"field":"copies","message":"冊数は0以上にしてください"}]}

# PATCH:何も送らない(検証に違反しない)
PATCH /1  {}
  -> 200  {"id":1,"title":"吾輩は猫である","copies":3}

# IDが0(URIの一部の検証)
GET /0
  -> 400  {"message":"入力に誤りがあります","errors":[{"field":"id","message":"IDは1以上にしてください"}]}

# 正しい入力
POST /  {"title":"こころ","copies":4}
  -> 201  {"id":3,"title":"こころ","copies":4}
